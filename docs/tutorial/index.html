Displaying a Window

Display a window and load a background image.
We won't go into detailed explanations here.
We will explain things step-by-step.
For now, please save the following script as is to startup.tjs.
The line numbers on the left are not needed.
From here on, we will modify this file as we build.

class MainWindow extends Window {
	var baselayer;
	function MainWindow( width, height ) {
		super.Window();
		setInnerSize( width, height );

		baselayer = new Layer(this, null);
		baselayer.loadImages("image.jpg");
		baselayer.setSizeToImageSize();
		baselayer.visible = true;
		add(baselayer);
	}
}
var mainwin = new MainWindow(640,480);
mainwin.visible = true;

Displaying an Image at the Mouse Position

Let's display an image at the mouse position to serve as the player's ship.
First, we need a layer to display the image.
Line 3 adds a member variable representing the layer.
Lines 14 to 18 create the layer and load the image.
Lines 20 to 22 move the image according to the mouse position.

class MainWindow extends Window {
	var baselayer;
	var movelayer;
	function MainWindow( width, height ) {
		super.Window();
		setInnerSize( width, height );

		baselayer = new Layer(this, null);
		baselayer.loadImages("image.jpg");
		baselayer.setSizeToImageSize();
		baselayer.visible = true;
		add(baselayer);

		movelayer = new Layer(this, baselayer);
		movelayer.loadImages("circle.png");
		movelayer.setSizeToImageSize();
		movelayer.visible = true;
		add(movelayer);
	}
	function onMouseMove(x, y, shift) {
		movelayer.setPos( x, y );
	}
}
var mainwin = new MainWindow(640,480);
mainwin.visible = true;

Displaying Centered on the Mouse

In the previous script, the image is displayed with the mouse position as its top-left corner, making the player's ship appear awkward to the lower-right of the cursor.
Therefore, we adjust the position so that it aligns with the mouse position.
Subtracting half of the layer's width and height from the mouse position will display it centered on the cursor.

class MainWindow extends Window {
	// Omitted
	function onMouseMove(x, y, shift) {
		movelayer.setPos( x-movelayer.width/2, y-movelayer.height/2 );
	}
}

Firing Bullets on Click

Add a bullet layer in the same way as the player's ship.
This corresponds to line 4 and lines 22 to 26.
We don't want the bullet to display until it is fired, so set visible = false to keep it hidden.
Line 28 enables waiting for vertical sync (VSync).
Line 29 adds a handler to ContinuousHandler.
Lines 31 to 33 remove the handler from ContinuousHandler.
Lines 37 to 40 display the bullet at the clicked position when clicked.
Lines 41 to 47 handle moving the bullet.

class MainWindow extends Window {
	var baselayer;
	var movelayer;
	var bulletlayer;
	function MainWindow( width, height ) {
		super.Window();
		setInnerSize( width, height );
		mouseCursorState = mcsHidden;

		baselayer = new Layer(this, null);
		baselayer.loadImages("image.jpg");
		baselayer.setSizeToImageSize();
		baselayer.visible = true;
		add(baselayer);

		movelayer = new Layer(this, baselayer);
		movelayer.loadImages("circle.png");
		movelayer.setSizeToImageSize();
		movelayer.visible = true;
		add(movelayer);

		bulletlayer = new Layer(this, baselayer);
		bulletlayer.loadImages("bullet.png");
		bulletlayer.setSizeToImageSize();
		bulletlayer.visible = false;
		add(bulletlayer);

		waitVSync = true;
		System.addContinuousHandler(handler);
	}
	function finalize() {
		System.removeContinuousHandler(handler);
	}
	function onMouseMove(x, y, shift) {
		movelayer.setPos( x-movelayer.width/2, y-movelayer.height/2 );
	}
	function onClick(x, y) {
		bulletlayer.setPos( x-movelayer.width/2, y-movelayer.height/2 );
		bulletlayer.visible = true;
	}
	function handler(tick) {
		if( bulletlayer.top > -bulletlayer.height ) {
			bulletlayer.top -= 10;
		} else {
			bulletlayer.visible = false;
		}
	}
}
var mainwin = new MainWindow(640,480);
mainwin.visible = true;
